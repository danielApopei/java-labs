CREATE OR REPLACE FUNCTION make_waves(p_sir_caractere IN varchar2 , v_rezultat OUT varchar2)  -- si utilizare variaila OUT
RETURN varchar2 AS
   v_index INTEGER;
 --  v_rezultat varchar2(1000) := ''; --daca folosim parametrul OUT, nu mai redenumim variabila aici
BEGIN
    FOR v_index IN 1..length(p_sir_caractere) LOOP
        IF(v_index MOD 2 = 1)
        THEN
           v_rezultat := v_rezultat || UPPER(SUBSTR(p_sir_caractere,v_index,1));
        ELSE
           v_rezultat := v_rezultat || LOWER(SUBSTR(p_sir_caractere,v_index,1));
        END IF;
    END LOOP;
  
   return v_rezultat; --neaparat return, pt ca e functie
END;

--- vom manageria info despre un student cu un id dat: afisare informatii si stergere student dat
-- se poate si dupa nr matricol

CREATE OR REPLACE PACKAGE manager_facultate IS   -- tot ce e in antet e public, variabila g_today_date va fi apelata din bloc anonim
    g_today_date   DATE:= SYSDATE;
    PROCEDURE sterge_student (p_id studenti.id%type);
    Procedure afiseaza_student (p_id studenti.id%type);
END manager_facultate;
/
--BODY--------
CREATE OR REPLACE PACKAGE BODY manager_facultate IS

    nume_facultate VARCHAR2(100) := 'Facultatea de Informatica din IASI';   -- variabila privata
    
    Function exista_student(p_id studenti.id%type )return boolean is    -- functie privata
    numar number;
    begin
        select count(*) into numar from studenti where id=p_id;
        if numar <>0 then
            return true;
        else return false;
        end if;
    end exista_student; --nu este obligatoriu sa se scrie numele functiei, dar ar fi indicat
    
    FUNCTION calculeaza_varsta(data_nastere DATE) RETURN INT AS   -- functie privata
    BEGIN
       RETURN FLOOR((g_today_date - data_nastere)/365);  -- apelam variabila publica
    END calculeaza_varsta;   

    Procedure afiseaza_student(p_id studenti.id%type) is
    v_nume studenti.nume%type;
    v_prenume studenti.prenume%type;
    v_matricol studenti.nr_matricol%type;
    v_nastere studenti.data_nastere%type;
    begin
        if exista_student(p_id) then  
            select nume, prenume, nr_matricol, data_nastere into v_nume, v_prenume, v_matricol, v_nastere from studenti where id = p_id;
            DBMS_OUTPUT.PUT_LINE(v_nume||' '||v_prenume||' '||v_matricol||' '||v_nastere||' '||calculeaza_varsta(v_nastere)  --apel fuct. privata
                                    ||' student la '||nume_facultate);
        else DBMS_OUTPUT.PUT_LINE(' Nu exista studentul cu id-ul ' ||p_id);
        end if;
    end;
    
    PROCEDURE sterge_student (p_id studenti.id%type) IS
    nr_prieteni number;
    nr_note number;
    BEGIN
        if exista_student(p_id) then
            select count(*) into nr_prieteni from prieteni where id_student1=p_id or id_student2 = p_id;
            if nr_prieteni <> 0 then delete from prieteni where id_student1=p_id or id_student2=p_id;
            end if;
      
            select count(*) into nr_note from note where id_student=p_id ;
            if nr_note <> 0 then delete from note where id_student=p_id;
            end if;
            
            delete from studenti where id = p_id;
            DBMS_OUTPUT.PUT_LINE('S-a sters studentul cu id-ul '||p_id);
        else DBMS_OUTPUT.PUT_LINE(' Nu exista studentul cu id-ul '||p_id||' ca sa il putem sterge');
        end if;
    END sterge_student;
    
END manager_facultate;
